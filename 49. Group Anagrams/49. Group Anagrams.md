    def groupAnagrams(self, strs: List[str]) -> List[List[str]]:
        anagram_dict = defaultdict(list)
        for s in strs:
            sorted_s = "".join(sorted(s))
            anagram_dict[sorted_s].append(s)
        res = []

度数分布自体をキーにするのはタプルを毎回生成するのがにオーバーヘッドでメモリを食いそうという先入観があって選択肢から無意識に外してるな。アルファベット小文字のようなUnicode上で連続している文字だけを扱っている場合はタプルでいいのだが、大文字も入れたい。さらに大文字と小文字を区別したいとなったときにタプルを使い続けるのはわかりにくいなどの気持ちもある。辞書や配列はmutableだからkeyにはなれないのでタプルを使っている。
ord(c) - ord('a')というのは度数分布にするのは便利だけどあんまり応用が効かないから個人的には好みでない。ただ値がマイナスになった時、範囲外になってクラッシュせずに動き続けてしまうのは盲点だった。

##  STEP 3
両方とも3回ずつ練習。
anagramが短ければソートでも頻度分布でもあまりでなさそうで無さそうだが単語の長さをkとしたときにソートだとO(klogk)頻度分布だとO(k)なので長くなると差がつくな。

sorted()とsort()がどっちがどっちだったかいつも混乱する。sortはlistの関数で元のlistをソートしてNoneを返すが、sortedはlist以外のiterableもとりソートされたリストを返す。

    def groupAnagrams(self, strs: List[str]) -> List[List[str]]:

                count[ord(c) - ord('a')] += 1
            freq_to_anagrams[tuple(count)].append(s)
        result = []
        for anagrams in freq_to_anagrams.values():
            result.append(anagrams)
        return result
