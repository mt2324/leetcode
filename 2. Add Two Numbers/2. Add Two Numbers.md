## STEP 1
一桁ずつ足して、carryとあまりを計算するして新しいノードを作りつなげていく。dummyを使うといい感じになりそう。

```python
class Solution:
    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
        carry = 0
        dummy = ListNode()
        node = dummy
        while l1 or l2:
            l1_val = l1.val if l1 else 0
            l2_val = l2.val if l2 else 0
            node.next = ListNode((l1_val + l2_val + carry) % 10)
            carry = (l1_val + l2_val + carry) // 10
            node = node.next
            l1 = l1.next if l1 else None
            l2 = l2.next if l2 else None
        if carry:
            node.next = ListNode(val=carry)
        return dummy.next
        
```

```python
class Solution:
    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
        carry = 0
        dummy = ListNode()
        node = dummy
        while l1 is not None or l2 is not None or carry != 0:
            l1_val = l1.val if l1 is not None else 0
            l2_val = l2.val if l2 is not None else 0
            carry, remainder = divmod(l1_val + l2_val + carry, 10)
            node.next = ListNode(val=remainder)
            l1 = l1.next if l1 is not None else None
            l2 = l2.next if l2 is not None else None
            node = node.next
        return dummy.next
```
