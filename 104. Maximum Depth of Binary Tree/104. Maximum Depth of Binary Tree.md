これはボトムアップ的なアプローチ。下の高さを聞いてくる。

# Definition for a binary tree node.

#     def __init__(self, val=0, left=None, right=None):

class Solution:
    def maxDepth(self, root: Optional[TreeNode]) -> int: 
        if root is None:
            return 0
        return max(self.maxDepth(root.left), self.maxDepth(root.right)) + 1

## STEP 2

### BFSの書き方色々
各層のサイズを渡すパターン
```python
from collections import deque


# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def maxDepth(self, root: Optional[TreeNode]) -> int: 
        if root is None:
            return 0
        nodes_to_visit = deque()
        nodes_to_visit.append(root)
        depth = 0
        while nodes_to_visit:
            level_size = len(nodes_to_visit)
            depth += 1
            for i in range(level_size):
                node = nodes_to_visit.popleft()
                if node.right is not None:
                    nodes_to_visit.append(node.right)
                if node.left is not None:
                    nodes_to_visit.append(node.left)
        return depth
```

次の層のキューを用意しておくパターン
```python
from collections import deque


# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def maxDepth(self, root: Optional[TreeNode]) -> int:
        if root is None:
            return 0
        same_level_nodes = deque()
        same_level_nodes.append(root)
        next_level_nodes = deque()
        max_depth = 0
        while same_level_nodes:
            node = same_level_nodes.pop()
            if node.right is not None:
                next_level_nodes.append(node.right)
            if node.left is not None:
                next_level_nodes.append(node.left)
            if not same_level_nodes:
                same_level_nodes = next_level_nodes
                next_level_nodes = deque()
                max_depth += 1
        return max_depth
```

### DFS
dfsにはトップダウン的なアプローチとボトムアップ的なアプローチがあり、それぞれ再帰関数を使って書いたりstackを使って書くこともできる。私の感覚ではボトムアップの方が再帰関数でトップダウンの方がstackを使って書くことが自然にできるけれども逆は難しいので練習が必要。backtrackingはトップダウン方のdfsを再帰関数を使って書いてるというイメージに近い。

dfs トップダウン 再帰
```python
class Solution:
    def maxDepth(self, root: Optional[TreeNode]) -> int:
        max_depth = 0
        
        def compute_depth(node, current_depth):
            nonlocal max_depth
            max_depth = max(max_depth, current_depth)
            if node.right is not None:
                compute_depth(node.right, current_depth + 1)
            if node.left is not None:
                compute_depth(node.left, current_depth + 1)
                
        if root is not None:
            compute_depth(root, 1)
        return max_depth
```

ボトムアップのdfsでstackを使った場合にはvisitedを管理するもしくは2つのstackを使うパターンがある。
visitedを管理する場合。
```python
def maxDepth(root) -> int:
    if root is None:
        return 0
    
    stack = [(root, False)]
    depths = {}  # 各ノードの深さを保持する辞書（ボトムアップの「答えの持ち帰り」を模倣）
    max_depth = 0
    
    while stack:
        node, visited = stack.pop()
        
        if visited:
            # 【ボトムアップのタイミング】子たちの答えが出揃ったあとの処理
            left_depth = depths.get(node.left, 0)
            right_depth = depths.get(node.right, 0)
            current_depth = max(left_depth, right_depth) + 1
            depths[node] = current_depth
            max_depth = max(max_depth, current_depth)
        else:
            # 【行きがけ】まだ子を処理していないので、
            # 「自分（Visited=True）」を積み、そのあとに「右・左の子（Visited=False）」を積む
            stack.append((node, True))
            if node.right is not None:
                stack.append((node.right, False))
            if node.left is not None:
                stack.append((node.left, False))
                    
    return max_depth
```

stack2個使う
```python
class Solution:
    def maxDepth(self, root: Optional[TreeNode]) -> int:
        if root is None:
            return 0
        
        # スタック1：探索用、スタック2：ポストオーダーの順序を作る用
        stack = [root]
        post_order_stack = []
        
        while stack:
            node = stack.pop()
            post_order_stack.append(node)
            
            # 左右の子をスタック1に入れる
            # （ポップしたときに左が先に処理されるよう、左を先に積む）
            if node.left is not None:
                stack.append(node.left)
            if node.right is not None:
                stack.append(node.right)
                
        # 各ノードの深さを記録する辞書（ボトムアップの値を保持）
        depths = {}
        max_depth = 0
        
        # post_order_stack を逆から取り出すと「左 -> 右 -> 根（ポストオーダー）」になる
        while post_order_stack:
            node = post_order_stack.pop()
            
            left_depth = depths.get(node.left, 0)
            right_depth = depths.get(node.right, 0)
            
            # ボトムアップに自分の深さを決定
            current_depth = max(left_depth, right_depth) + 1
            depths[node] = current_depth
            
            max_depth = max(max_depth, current_depth)
            
        return max_depth

```

        max_depth = 0

            max_depth = max(max_depth, depth)

dfsのボトムアップをstackで書くのを練習。この問題に対しては複雑すぎる回答だが、練習は必要
stack 1個で方向を渡すバージョン
```python
class Solution:
    def maxDepth(self, root: Optional[TreeNode]) -> int:
        if root is None:
            return 0
        stack = []
        stack.append((root, False))
        max_depth = 0
        depths = {}
        while stack:
            node, visited = stack.pop()
            if visited:
                left_depth = depths.get(node.left, 0)
                right_depth = depths.get(node.right, 0)
                depth = max(left_depth, right_depth) + 1
                depths[node] = depth
                max_depth = max(max_depth, depth)
            else:
                stack.append((node, True))
                if node.right is not None:
                    stack.append((node.right, False))
                if node.left is not None:
                    stack.append((node.left, False))
        return max_depth
```

行きのstack帰りのstackを分けるバージョン。こちらの方が条件分岐が減るのでまだこちらの方が好みではある。
```python
class Solution:
    def maxDepth(self, root: Optional[TreeNode]) -> int:
        if root is None:
            return 0
        go_stack = []
        back_stack = []
        go_stack.append(root)
        while go_stack:
            node = go_stack.pop()
            back_stack.append(node)
            if node.right is not None:
                go_stack.append(node.right)
            if node.left is not None:
                go_stack.append(node.left)
        depths = {}
        max_depth = 0
        while back_stack:
            node = back_stack.pop()
            left_depth = depths.get(node.left, 0)
            right_depth = depths.get(node.right, 0)
            depth = max(left_depth, right_depth) + 1
            depths[node] = depth
            max_depth = max(depth, max_depth)
        return max_depth   
```
